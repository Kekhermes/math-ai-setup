\begin{document}
\maketitle

\begin{proposition}[Variante A — Anneau commutatif $\Z$]
Pour tous entiers $a, b \in \Z$, on a :
\[
  (a + b)^2 = a^2 + 2ab + b^2.
\]
\end{proposition}

\begin{proposition}[Variante B — Matrices carrées $\mathcal{M}_2(\Z)$ sans hypothèse]
Pour toutes matrices $A, B \in \mathcal{M}_2(\Z)$, on aurait :
\[
  (A + B)^2 = A^2 + 2AB + B^2.
\]
\end{proposition}
